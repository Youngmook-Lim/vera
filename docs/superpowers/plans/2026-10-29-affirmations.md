**Goal:** Slot 04 becomes AFFIRMATIONS: Louise Hay's affirmations from her own site, one poster at a time on a full field of colour, swiped through, with favourites, a list of all of them, and a daily card at 10:00.

**Architecture:** A room in the advice room's mould — `ModuleDef` in `src/modules/affirmations/`, its own store, `stored.ts`, `strings.ts`, `notify.ts` on the shared `makeDaily`, the shared `deck.ts` and `favorites.ts` — with a new look and a new gesture. **Tech Stack:** React Native 0.86 + Expo SDK 57, TypeScript, Zustand, react-native-reanimated 4.5, expo-notifications (mocked in jest), Jest 29 + RTL v14 (fully async).

**Spec:** `docs/superpowers/specs/2026-10-29-affirmations-design.md`

## Global Constraints

- Every field/type pair holds **4.5:1**.
- Commit messages carry no reference to Claude or the model (no Co-Authored-By, no Claude-Session line).

// One card back, for a room you swipe through (module 04). It never crosses
// the seam: the previous pass's order was thrown away when `advance` dealt a
// new one, so the first card of a pass is where going back stops. It hands
// back the SAME deck there rather than an equal one, which is how a caller
// tells that nothing moved.
export const retreat = (deck: Deck): Deck =>
  deck.at === 0 ? deck : { order: deck.order, at: deck.at - 1 };

```js
// Builds `src/modules/affirmations/lines.ts` from Louise Hay's own site.
//
//   node tools/build-affirmations.mjs
//
// Two pages on louisehay.com publish her affirmations verbatim: a numbered
// list of 101, and a gallery whose every image carries its line in the alt
// text. This reads both, keeps the 101-best page's order first and appends
// the gallery's lines it does not already have, and drops exact duplicates.
// Near-duplicates stay: they are her own two wordings, and only identity can be
// decided by a script.
//
// The output is REVIEWED BY HAND before it is committed, because these are web
// pages and not a book. Anything that is a caption rather than an affirmation
// goes into EXCLUDE below, never into a hand-edit of the output, so running
// this again reproduces the file exactly. The order is pinned by
// lines.test.ts: a card's numeral is its position here.
//
// Her lines are still under copyright. This is a personal build, on the same
// footing as the horror room's corpus.

import { writeFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';

const OUT = 'src/modules/affirmations/lines.ts';
const BEST = 'https://www.louisehay.com/101-best-louise-hay-positive-affirmations/';
const GALLERY = 'https://www.louisehay.com/affirmations/';

// Lines either page carries that are not affirmations. Empty until a review
// finds one.
const EXCLUDE = new Set([]);

const ENTITIES = { amp: '&', quot: '"', apos: "'", lt: '<', gt: '>', nbsp: ' ' };
const unescape = (s) =>
  s
    .replace(/&#(\d+);/g, (_, n) => String.fromCodePoint(Number(n)))
    .replace(/&#x([0-9a-f]+);/gi, (_, n) => String.fromCodePoint(parseInt(n, 16)))
    .replace(/&([a-z]+);/gi, (m, n) => ENTITIES[n.toLowerCase()] ?? m);
const clean = (s) => unescape(s.replace(/<[^>]+>/g, '')).replace(/\s+/g, ' ').trim();

const get = async (url) => {
  const res = await fetch(url, { headers: { 'user-agent': 'Mozilla/5.0' } });
  if (!res.ok) throw new Error(`${url}: http ${res.status}`);
  return res.text();
};

// The first <ol> after "Let's affirm:" is the list. Some items are wrapped in
// <strong>; the tags go and the words stay.
const fromBest = (html) => {
  const start = html.indexOf('<ol>', html.indexOf('affirm:'));
  if (start < 0) throw new Error('101-best: no list after "affirm:"');
  const list = html.slice(start, html.indexOf('</ol>', start));
  return [...list.matchAll(/<li>([\s\S]*?)<\/li>/g)].map((m) => clean(m[1]));
};

// Each alt reads `Inspirational Quotes about <topic> | "<line>" — Louise Hay`.
// One that has a topic but does not parse is reported rather than guessed at.
const fromGallery = (html) => {
  const out = [];
  for (const [, raw] of html.matchAll(/alt="([^"]*)"/g)) {
    const alt = unescape(raw);
    const m = alt.match(/\|\s*"([\s\S]*)"\s*[—-]\s*Louise Hay\s*$/);
    if (m) out.push(clean(m[1]));
    else if (alt.includes('|')) console.warn(`skipped: ${alt}`);
  }
  return out;
};

const quote = (s) => `'${s.replace(/\\/g, '\\\\').replace(/'/g, "\\'")}'`;

const seen = new Set();
const lines = [];
for (const line of [...fromBest(await get(BEST)), ...fromGallery(await get(GALLERY))]) {
  if (!line || EXCLUDE.has(line) || seen.has(line)) continue;
  seen.add(line);
  lines.push(line);
}

const body = `// GENERATED by tools/build-affirmations.mjs - do not hand-edit a line.
//
// ${lines.length} affirmations by Louise Hay, verbatim from her own site: the
// 101-best page in its own order, then the gallery's lines it did not already
// have. A card's numeral is its position here, so the order is pinned by
// lines.test.ts and re-running the script must not move a line.
//
// Still under copyright. A personal build, the same footing as the horror
// room's corpus; this file would be replaced before anything shipped.

export const SOURCE = {
  author: 'Louise Hay',
  urls: [
    ${quote(BEST)},
    ${quote(GALLERY)},
  ],
} as const;

export const LINES: readonly string[] = [
${lines.map((l) => `  ${quote(l)},`).join('\n')}
];
`;

mkdirSync(dirname(OUT), { recursive: true });
writeFileSync(OUT, body);
console.log(`${lines.length} lines -> ${OUT}`);
```

  // Keyed by the line's own words, never its position. A re-run of the script
  // that moved a line must not move a favourite with it.
  favorites: Favorites;
  daily: boolean;
  syncedOn: string | null;
}

export const EMPTY_STORED: Stored = {
  favorites: new Set<string>(),
  daily: false,
  syncedOn: null,
};

interface Blob {
  favorites: string[];
  daily: boolean;
  syncedOn: string | null;
}

export const serialize = (s: Stored): string =>
  JSON.stringify({
    favorites: [...s.favorites],
    daily: s.daily,
    syncedOn: s.syncedOn,
  } satisfies Blob);

// Anything that is not the shape expected is that field absent, never a throw.
export const parse = (raw: string | null): Stored => {
  if (!raw) return { ...EMPTY_STORED, favorites: new Set<string>() };
  try {
    const blob = JSON.parse(raw) as Partial<Blob>;
    return {
      favorites: new Set(
        Array.isArray(blob.favorites)
          ? blob.favorites.filter((q): q is string => typeof q === 'string')
          : [],
      ),
      daily: blob.daily === true,
      syncedOn: typeof blob.syncedOn === 'string' ? blob.syncedOn : null,
    };
  } catch {
    return { ...EMPTY_STORED, favorites: new Set<string>() };
  }
};

```ts
import { useEffect } from 'react';
import { create } from 'zustand';
import type { Lang } from '../../core/i18n';
import { KEYS, storage } from '../../core/storage';
import { dueForSync } from '../daily';
import { type Deck, newDeck } from '../deck';
import { type Favorites, toggle } from '../favorites';
import { LINES } from './lines';
import { requestDaily, syncDaily } from './notify';
import { type Stored, parse, serialize } from './stored';

// The cards you kept, owned by the room that shows them. She is not told this
// exists. The advice room's store, line for line, because the bargain is the
// same: a failed write costs this session's marks and never reaches a void
// call site as an unhandled rejection.
const save = async (stored: Stored): Promise<void> => {
  try {
    await storage.set(KEYS.affirmations, serialize(stored));
  } catch {
    // an unwritable disk — the card is kept for this session only
  }
};

interface AffirmationsState extends Stored {
  hydrated: boolean;
  // The pass. Here rather than in the screen so walking out and back in does
  // not reshuffle; NOT in `Stored`, so a restart does.
  deck: Deck;
  setDeck: (next: (deck: Deck) => Deck) => void;
  hydrate: () => Promise<void>;
  toggleFavorite: (line: string) => Promise<void>;
  setDaily: (on: boolean, lang: Lang) => Promise<boolean>;
  syncIfDue: (lang: Lang, now?: Date) => Promise<void>;
}

let inFlight: Promise<void> | null = null;

export const useAffirmationsStore = create<AffirmationsState>((set, get) => ({
  hydrated: false,
  deck: newDeck(LINES.length),
  setDeck: (next) => set((s) => ({ deck: next(s.deck) })),
  favorites: new Set<string>(),
  daily: false,
  syncedOn: null,

  hydrate: async (): Promise<void> => {
    if (get().hydrated) return;
    if (inFlight) return inFlight;

    inFlight = (async () => {
      let raw: string | null = null;
      try {
        raw = await storage.get(KEYS.affirmations);
      } catch {
        // unreadable disk — the room opens with nothing kept
      }
      set({ hydrated: true, ...parse(raw) });
    })();

    await inFlight;
    inFlight = null;
  },

  toggleFavorite: async (line: string): Promise<void> => {
    await get().hydrate();
    const favorites = toggle(get().favorites, line);
    set({ favorites });
    await save({ ...get(), favorites });
  },

  // Returns whether it is on afterwards, which is NOT what was asked for when
  // the permission is refused: Android never re-prompts.
  setDaily: async (on: boolean, lang: Lang): Promise<boolean> => {
    // SYSTEM can reach this before the room has mounted; without the hydrate,
    // `save` would write an empty set over every kept card.
    await get().hydrate();
    if (on && !(await requestDaily())) {
      return false;
    }
    const now = new Date();
    // Two facts: `daily` is what was asked for, `syncedOn` is whether Android
    // took it. The stamp lands only on success.
    set({ daily: on });
    const ok = await syncDaily(on, lang, now);
    const syncedOn = on && ok ? now.toDateString() : null;
    set({ syncedOn });
    await save({ ...get(), daily: on, syncedOn });
    return on;
  },

  syncIfDue: async (lang: Lang, now: Date = new Date()): Promise<void> => {
    const { daily, syncedOn } = get();
    if (!daily) return;
    if (!dueForSync(syncedOn, now)) return;
    if (!(await syncDaily(true, lang, now))) return;

    const stamp = now.toDateString();
    set({ syncedOn: stamp });
    await save({ ...get(), syncedOn: stamp });
  },
}));

// Everything that shows a star goes through this: it subscribes to the live
// set and makes sure something has read the disk first.
export const useFavorites = (): {
  favorites: Favorites;
  toggleFavorite: (line: string) => Promise<void>;
} => {
  const favorites = useAffirmationsStore((s) => s.favorites);
  const toggleFavorite = useAffirmationsStore((s) => s.toggleFavorite);
  const hydrate = useAffirmationsStore((s) => s.hydrate);

  useEffect(() => {
    void hydrate();
  }, [hydrate]);

  return { favorites, toggleFavorite };
};
```

```ts
// Every decision the poster's swipe makes, pure and out of the gesture, because
// a worklet and a PanResponder are both unreachable from jest (the stage's
// dock button taught that a mutation inside one survives the whole suite).

export type Swipe = 'next' | 'prev' | 'stay';

// Past this share of the screen a release moves on however slowly it ended.
export const SWIPE_FRACTION = 0.25;
// Or a release this fast, in PanResponder's own unit (px per ms), in the
// direction of the drag.
export const SWIPE_VELOCITY = 0.5;
// Below this a drag is a touch that wobbled, not a swipe.
const CLAIM_PX = 10;

// Whether the poster takes the drag from whatever is under it — the
// ScrollView a long line sits in. Only clearly sideways drags.
export const claims = (dx: number, dy: number): boolean =>
  Math.abs(dx) > CLAIM_PX && Math.abs(dx) > Math.abs(dy);

export const settle = (dx: number, vx: number, width: number): Swipe => {
  if (dx === 0) return 'stay';
  const far = Math.abs(dx) >= width * SWIPE_FRACTION;
  const flung = Math.sign(vx) === Math.sign(dx) && Math.abs(vx) >= SWIPE_VELOCITY;
  if (!far && !flung) return 'stay';
  // Left is forward: the next card comes in from the right.
  return dx < 0 ? 'next' : 'prev';
};
```

  // back. Every decision is in swipe.ts, where a test can reach it.

  widthRef.current = width;
  const tx = useSharedValue(0);
  const slide = useAnimatedStyle(() => ({ transform: [{ translateX: tx.value }] }));

  // Reads the deck from the store rather than from this render, because the
  // responder below is built once and would otherwise hold the first deck
  // forever. `retreat` hands back the same deck at the start of a pass, which
  // is how a swipe back there knows to spring rather than slide.
  const go = (to: Swipe, dx = 0): void => {
    const was = useAffirmationsStore.getState().deck;
    const next = to === 'next' ? advance(was) : to === 'prev' ? retreat(was) : was;
    if (next === was) {
      tx.value = withSpring(0);
      return;
    }
    setDeck(() => next);
    // The new card enters from the side the old one left by, from wherever
    // the finger let go, so the swipe does not jump.
    const from = (to === 'next' ? widthRef.current : -widthRef.current) + dx;
    tx.value = withSequence(withTiming(from, { duration: 0 }), withTiming(0, { duration: SLIDE_MS }));
