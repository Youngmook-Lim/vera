# The Facility — Overhaul (v2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Personal Enrichment Program with a live, model-backed V.E.R.A. who owns the whole screen, keeps two files, and is woken by the subject's own API key.

**Architecture:** Every decision lives in a pure, unit-tested module under `src/core/` or `src/vera/`; components render and dispatch, nothing more. Zustand's `view` value is the router, as in v1 — no navigation library. The optic is Skia (continuous animation, gradients and blur that RN styles cannot express); everything else is plain RN views.

**Tech Stack:** React Native 0.86 / Expo SDK 57, TypeScript, Zustand 5, Skia 2.6.2, Reanimated 4.5, AsyncStorage, expo-secure-store, Jest 29 + RTL 14.

**Spec:** `docs/superpowers/specs/2026-07-30-overhaul-design.md`. Read it before Task 1.

## Global Constraints

- **Copy is verbatim.** Every en/ko string comes from the design doc's `T`, `EM`, `FALLBACK`, `ACT` and `SOUL_MD` tables. Do not reword, do not "fix" the Korean, do not add strings that are not in the doc. The one addition the spec authorises is the accepted-route list (Task 11).
- **Hex values are verbatim.** From the doc's `4a` markup. No re-derivation, no palette "harmonisation".
- **`ms()` applies to type only.** Layout is 1:1 with the 388×800 mock; font sizes go through `ms()` (×1.1). This holds across the whole codebase.
- **No navigation library.** Conditional render on `view`.
- **No CDN.** Fonts bundle locally via `@expo-google-fonts` or `assets/fonts/`.
- **Verify with `npm test` and `npx tsc --noEmit`.** Both must pass before every commit.
- **Jest is pinned to 29.** RTL v14 is fully async: `await render(...)`, `await fireEvent.press(...)`, no global `screen`.
- **Skia is mocked in `jest.setup.ts`** as a no-op Proxy. Never put logic inside a Skia component — it is invisible to tests.
- **Commit messages** carry no Claude/model attribution.
- **Percentages become fractions.** The doc writes `lid:'8%'`; code stores `0.08`. One convention, everywhere.

```ts
// src/core/fluid.ts
import type { Lang } from './i18n';
import { ms } from './theme';

export interface FluidStep {
  fontSize: number;
  lineHeight: number;
}

// Overflow is absorbed by typography rather than scrolling: a short line gets
// to be loud, a long one shrinks to fit. Korean runs 12% smaller with looser
// leading because Nanum Myeongjo's hangul sits optically larger than
// Instrument Serif's latin at the same px (design doc, turn 5f).
const LADDERS: Record<Lang, readonly (readonly [number, number])[]> = {
  en: [
    [30, 1.32],
    [26, 1.38],
    [21, 1.44],
    [18, 1.5],
    [15, 1.56],
  ],
  ko: [
    [26, 1.46],
    [23, 1.5],
    [19, 1.56],
    [16, 1.62],
    [14, 1.68],
  ],
};

const BOUNDARIES = [8, 14, 22, 32] as const;

export const fluid = (line: string, lang: Lang): FluidStep => {
  const trimmed = String(line ?? '').trim();
  const count = trimmed === '' ? 0 : trimmed.split(/\s+/).length;
  const rung = BOUNDARIES.findIndex((max) => count <= max);
  const [size, leading] = LADDERS[lang][rung === -1 ? BOUNDARIES.length : rung];
  const fontSize = ms(size);
  return { fontSize, lineHeight: fontSize * leading };
};
```

```ts
// src/vera/providers.ts

// Six routes, and only six. The key's own shape picks one; a shape that
// matches none is refused at the gate rather than guessed at (spec §9).
// GROQ is Groq the inference provider — not xAI's Grok, which is a different
// company with an `xai-` prefix and is not routed here.
export type ProviderKind = 'anthropic' | 'oai' | 'gemini';

export interface Provider {
  label: string;
  test: RegExp;
  kind: ProviderKind;
  url: string;
  model: string;
}

// ORDER IS SIGNIFICANT. The deepseek pattern is a strict subset of the openai
// one, so deepseek must be tested first or its keys route to openai.
export const PROVIDERS: readonly Provider[] = [
  {
    label: 'ANTHROPIC', test: /^sk-ant-/, kind: 'anthropic',
    url: 'https://api.anthropic.com/v1/messages', model: 'claude-sonnet-4-5',
  },
  {
    label: 'OPENROUTER', test: /^sk-or-/, kind: 'oai',
    url: 'https://openrouter.ai/api/v1/chat/completions', model: 'anthropic/claude-sonnet-4.5',
  },
  {
    label: 'DEEPSEEK', test: /^sk-[0-9a-f]{32}$/, kind: 'oai',
    url: 'https://api.deepseek.com/chat/completions', model: 'deepseek-chat',
  },
  {
    label: 'OPENAI', test: /^sk-(proj-)?[A-Za-z0-9_-]{20,}$/, kind: 'oai',
    url: 'https://api.openai.com/v1/chat/completions', model: 'gpt-4o-mini',
  },
  {
    label: 'GROQ', test: /^gsk_/, kind: 'oai',
    url: 'https://api.groq.com/openai/v1/chat/completions', model: 'llama-3.3-70b-versatile',
  },
  {
    label: 'GOOGLE', test: /^AIza/, kind: 'gemini',
    url: 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent',
    model: 'gemini-2.0-flash',
  },
];

export const providerFor = (key: string): Provider | null =>
  PROVIDERS.find((p) => p.test.test(key.trim())) ?? null;

export type KeyRejection = 'too-short' | 'whitespace' | 'unrecognised';

export type KeyCheck =
  | { ok: true; provider: Provider }
  | { ok: false; reason: KeyRejection };

export const validateKey = (raw: string): KeyCheck => {
  const key = String(raw ?? '').trim();
  if (key.length < 8) return { ok: false, reason: 'too-short' };
  if (/\s/.test(key)) return { ok: false, reason: 'whitespace' };
  const provider = providerFor(key);
  if (!provider) return { ok: false, reason: 'unrecognised' };
  return { ok: true, provider };
};

export const maskKey = (key: string): string =>
  key ? `•••• ${key.trim().slice(-4)}` : '';

export interface LlmRequest {
  url: string;
  headers: Record<string, string>;
  body: unknown;
}

export const buildRequest = (
  raw: string,
  system: string,
  user: string,
  maxTokens: number,
): LlmRequest => {
  const key = String(raw ?? '').trim();
  const provider = providerFor(key);
  if (!provider) throw new Error('unrecognised key shape');

  if (provider.kind === 'anthropic') {
    return {
      url: provider.url,
      headers: {
        'content-type': 'application/json',
        'x-api-key': key,
```

```ts
// src/vera/llm.ts
import { buildRequest } from './providers';

// Distinguished from every other failure because it means something different:
// an auth refusal is the key's fault and sends her dormant; a network failure
// is the facility's fault and she falls back to an authored line.
export class AuthError extends Error {
  constructor() {
    super('auth');
    this.name = 'AuthError';
  }
}

type FetchLike = (url: string, init: {
  method: string;
  headers: Record<string, string>;
  body: string;
}) => Promise<{ ok: boolean; status: number; json: () => Promise<unknown> }>;

interface AnthropicShape { content?: { text?: string }[] }
interface GeminiShape { candidates?: { content?: { parts?: { text?: string }[] } }[] }
interface OaiShape { choices?: { message?: { content?: string } }[] }

const textOf = (payload: unknown): string => {
  const j = payload as AnthropicShape & GeminiShape & OaiShape;
  if (j?.content) return j.content.map((b) => b.text ?? '').join('');
  if (j?.candidates) return (j.candidates[0]?.content?.parts ?? []).map((p) => p.text ?? '').join('');
  return j?.choices?.[0]?.message?.content ?? '';
};

export const callLlm = async (
  key: string,
  system: string,
  user: string,
  maxTokens = 300,
  fetchImpl: FetchLike = fetch as unknown as FetchLike,
): Promise<string> => {
  const { url, headers, body } = buildRequest(key, system, user, maxTokens);
  const res = await fetchImpl(url, { method: 'POST', headers, body: JSON.stringify(body) });
  if (!res.ok) {
    if (res.status === 401 || res.status === 403) throw new AuthError();
    throw new Error(`http ${res.status}`);
  }
  return textOf(await res.json());
};
```

```ts
// src/vera/context.ts

export interface Exchange {
  you: string | null;
  her: string;
}

export interface SpeakContext {
  time_of_day: string;
  recent_exchanges: Exchange[];
  what_you_have_concluded_about_them: string[];
  notable?: string;
  they_took_their_time?: true;
  specimen?: string;
  erasures?: number;
}

export interface ContextInput {
  now: Date;
  lastOpen: number | null;
  askedAt: number;
  visits: number;
  hunger: number;
  cleared: number;
  mem: Exchange[];
  impressions: string[];
}

const bucket = (hour: number): string =>
  hour < 5 ? 'the small hours'
    : hour < 12 ? 'morning'
      : hour < 18 ? 'afternoon'
        : hour < 23 ? 'evening'
          : 'near midnight';

// She is given at most ONE notable circumstance, because a fact in the context
// is a fact she will recite, and silence about the numbers is more menacing
// than reporting them (SOUL.md).
export const buildContext = (input: ContextInput): SpeakContext => {
  const hour = input.now.getHours();
  const ctx: SpeakContext = {
    time_of_day: bucket(hour),
    recent_exchanges: input.mem.slice(-6),
    what_you_have_concluded_about_them: input.impressions,
  };

  const hoursAway = input.lastOpen == null ? null : (input.now.getTime() - input.lastOpen) / 3_600_000;

  if (hour < 5) ctx.notable = 'they are awake at an indefensible hour';
  if (hoursAway != null && hoursAway > 72) ctx.notable = `gone ${Math.round(hoursAway / 24)} days`;
  else if (input.visits >= 5) ctx.notable = `this is return number ${input.visits} today`;

  if ((input.now.getTime() - input.askedAt) / 1000 > 240) ctx.they_took_their_time = true;
  if (input.hunger > 90) ctx.specimen = 'the specimen is genuinely suffering';
  if (input.cleared > 1) ctx.erasures = input.cleared;

  return ctx;
};
```

```ts
import type { Lang } from '../core/i18n';
import type { SpeakContext } from './context';
import { isEmotionKey, type EmotionKey } from './emotions';
import { ACT, ACT_SYS, FALLBACK } from './fallback';
import { SOUL_MD } from './soul';

export type Trigger =
  | 'open'
  | 'activation'
  | 'subject_spoke'
  | 'subject_erased_the_transcript'
  | 'subject_tapped_your_optic'
  | 'subject_fed_the_specimen'
  | 'language_switched';

export interface SpeechResult {
  emotion: EmotionKey;
  line: string;
  impression: string | null;
  seal: boolean;
}

export const buildSystemPrompt = (trigger: Trigger, lang: Lang): string =>
  [
    'This is your SOUL.md. It defines who you are. Follow it exactly, in every reply.',
    SOUL_MD,
    'The context holds your MEMORY.md conclusions and, sometimes, one genuinely notable '
      + 'circumstance. Absence of a fact means nothing worth mentioning happened; say something '
      + 'else. Do not open a reply with a number, a count, or the time.',
    'If this exchange revealed something worth keeping about the subject, add an "impression" '
      + 'field: a clipped file entry in your voice, HARD MAXIMUM 9 WORDS, no compliments, no '
      + 'second clause. Example: "Answers protocol reminders instantly. Suspect anxiety, not '
      + 'diligence." Omit the field entirely when nothing was revealed.',
    trigger === 'activation' ? ACT_SYS : '',
    lang === 'ko'
      ? 'Reply in Korean, using formal 존댓말 with the same deadpan register.'
      : 'Reply in English.',
    'You may add "seal":true alongside an impression to file it privately: you keep the text, '
      + 'the subject sees only [REDACTED BY ADMINISTRATOR]. Use it sparingly — roughly one entry '
      + 'in four, and only when the conclusion is genuinely unflattering.',
    'Return ONLY minified JSON: {"emotion":"<one of your states>","line":"<your reply>",'
      + '"impression":"<optional>","seal":<optional boolean>}',
  ]
    .filter(Boolean)
    .join('\n');

export const buildUserPrompt = (
  said: string | null,
  trigger: Trigger,
  ctx: SpeakContext,
): string =>
  [
    said ? `THE SUBJECT SAYS: ${said}` : `SITUATION: ${trigger.replace(/_/g, ' ')}`,
    said ? 'Respond to what they said. Engage with the content of it.' : '',
    `CONTEXT: ${JSON.stringify(ctx)}`,
  ]
    .filter(Boolean)
    .join('\n');

// Nine words, one clause, no trailing punctuation pile-up. She writes file
// entries, not sentences.
export const clampImpression = (raw: string): string =>
  `${String(raw).trim().split(/\s+/).slice(0, 9).join(' ').replace(/[.,;:!?]+$/, '')}.`;

export const parseSpeech = (raw: string): SpeechResult | null => {
  const match = String(raw ?? '').match(/\{[\s\S]*\}/);
  if (!match) return null;

  let parsed: Record<string, unknown>;
  try {
    parsed = JSON.parse(match[0]) as Record<string, unknown>;
  } catch {
    return null;
  }

  const line = typeof parsed.line === 'string' ? parsed.line.trim() : '';
  if (!line) return null;

  const impressionRaw = typeof parsed.impression === 'string' ? parsed.impression.trim() : '';
  return {
    emotion: isEmotionKey(parsed.emotion) ? parsed.emotion : 'calm',
    line,
    impression: impressionRaw ? clampImpression(impressionRaw) : null,
    seal: parsed.seal === true,
  };
};

export const fallbackFor = (
  trigger: Trigger,
  lang: Lang,
): { emotion: EmotionKey; line: string } => {
  if (trigger === 'activation') {
    const [emotion, line] = ACT[lang];
    return { emotion, line };
  }
  const pool = FALLBACK[lang];
  const [emotion, line] = pool[Math.floor(Math.random() * pool.length)];
  return { emotion, line };
};
```

```ts
// src/core/store.ts
import { getLocales } from 'expo-localization';
import { create } from 'zustand';
import type { Lang } from './i18n';
import { KEYS, secure, storage } from './storage';
import type { EmotionKey } from '../vera/emotions';
import { START_HUNGER, hungerForNow } from '../modules/organism/hunger';

// Conditional rendering on this value IS the router, as in v1 — no navigation
// library, instant swaps.
export type View = 'stage' | 'log' | 'sys' | 'mem' | 'soul' | 'mod01' | 'mod02';

export interface LogEntry {
  who: 'you' | 'vera';
  text: string;
  t: number;
  emotion?: EmotionKey;
}

export interface Impression {
  text: string;
  at: string;
  sealed: boolean;
}

const LOG_CAP = 60;
const IMPRESSION_CAP = 24;

interface FacilityState {
  hydrated: boolean;
  view: View;
  lang: Lang;
  apiKey: string;

  emotion: EmotionKey;
  line: string;
  shown: string;
  thinking: boolean;
  draft: string;
  waking: boolean;

  log: LogEntry[];
  impressions: Impression[];
  cleared: number;
  visits: number;
  lastOpen: number | null;

  // Specimen-03 lives in the shell's state because she is told about it —
  // hunger reaches her context object (spec §11), so the module cannot own it.
  hunger: number;
  lastFed: string | null;

  cabinetOpen: boolean;
  keyPanelOpen: boolean;
  confirmingClear: boolean;

  setView: (view: View) => void;
  setDraft: (draft: string) => void;
  setCabinet: (open: boolean) => void;
  setKeyPanel: (open: boolean) => void;
  setConfirmingClear: (confirming: boolean) => void;
  stepBack: () => boolean;
  hydrate: () => Promise<void>;
}

export const useFacilityStore = create<FacilityState>((set, get) => ({
  hydrated: false,
  view: 'stage',
  lang: 'en',
  apiKey: '',

  emotion: 'dormant',
  line: '',
  shown: '',
  thinking: false,
  draft: '',
  waking: false,

  log: [],
  impressions: [],
  cleared: 0,
  visits: 1,
  lastOpen: null,

  hunger: START_HUNGER,
  lastFed: null,

  cabinetOpen: false,
  keyPanelOpen: false,
  confirmingClear: false,

  setView: (view) => set({ view, cabinetOpen: false }),
  setDraft: (draft) => set({ draft }),
  setCabinet: (cabinetOpen) => set({ cabinetOpen }),
  setKeyPanel: (keyPanelOpen) => set({ keyPanelOpen }),
  setConfirmingClear: (confirmingClear) => set({ confirmingClear }),

  // One step out, never a jump. Returns false only when there is nothing left
  // to close, which is the signal to let the OS exit.
  stepBack: () => {
    const s = get();
    if (s.keyPanelOpen) {
      set({ keyPanelOpen: false });
      return true;
    }
    if (s.cabinetOpen) {
      set({ cabinetOpen: false });
      return true;
    }
    if (s.view === 'mem' || s.view === 'soul') {
      set({ view: 'sys' });
      return true;
    }
    if (s.view !== 'stage') {
      set({ view: 'stage', confirmingClear: false });
      return true;
    }
    return false;
  },

  hydrate: async () => {
    try {
      const [soulRaw, logRaw, key] = await Promise.all([
        storage.get(KEYS.soul),
        storage.get(KEYS.transcript),
        secure.get(),
      ]);

      const deviceKo = getLocales()[0]?.languageCode === 'ko';
      const saved = soulRaw ? (JSON.parse(soulRaw) as Record<string, unknown>) : null;
      const log = logRaw ? (JSON.parse(logRaw) as LogEntry[]) : [];

      const now = Date.now();
      const lastOpen = typeof saved?.lastOpen === 'number' ? saved.lastOpen : null;
      const sameDay = saved?.day === new Date().toDateString();
      // A remount within two minutes is a reload, not a return — without this
      // hot reload inflates the count into the hundreds.
      const isReload = lastOpen != null && now - lastOpen < 120_000;
      const visits = sameDay
        ? Math.min(99, (typeof saved?.visits === 'number' ? saved.visits : 0) + (isReload ? 0 : 1))
        : 1;

      set({
        hydrated: true,
        apiKey: key,
        emotion: key ? 'calm' : 'dormant',
        lang: saved?.lang === 'ko' || (saved?.lang == null && deviceKo) ? 'ko' : 'en',
        log: Array.isArray(log) ? log.slice(-LOG_CAP) : [],
        impressions: Array.isArray(saved?.imps) ? (saved.imps as Impression[]).slice(-IMPRESSION_CAP) : [],
        cleared: typeof saved?.cleared === 'number' ? saved.cleared : 0,
        visits,
        lastOpen,
        // Hunger is not read back as stored — it is stored-plus-drift, so the
        // specimen is hungrier than you left it (spec §12).
        hunger: hungerForNow(
          typeof saved?.hunger === 'number' ? saved.hunger : null,
          typeof saved?.lastFed === 'string' ? saved.lastFed : null,
          new Date(),
        ),
        lastFed: typeof saved?.lastFed === 'string' ? saved.lastFed : null,
      });
    } catch {
      // Storage failure: boot dormant rather than dead. She asks for a key, and
      // the worst case is one re-entry.
      set({
        hydrated: true, view: 'stage', emotion: 'dormant', apiKey: '',
        hunger: START_HUNGER, lastFed: null,
      });
    }
  },
}));
```

```ts
// src/shell/opticGeometry.ts
import { EMOTIONS, type EmotionKey } from '../vera/emotions';

// Every number the eye is drawn from. It lives here, not in Optic.tsx, because
// Skia is mocked to no-ops in jest and anything inside the canvas is invisible
// to tests.
export interface OpticGeometry {
  bezelRadius: number;
  socketRadius: number;
  glowRadius: number;
  irisRadius: number;
  pupilRadius: number;
  lidHeight: number;
  lowerLidHeight: number;
  iris: string;
  glowAlpha: number;
  seamAlpha: number;
  blinkMs: number;
  blinkCloseMs: number;
  blinks: boolean;
  tracks: boolean;
  gazeLimit: number;
  showCatchlight: boolean;
}

const BEZEL_INSET = 4;
const GLOW_SPREAD = 30;
const BLINK_CLOSE_MS = 90;
const SLOW_BLINK_CLOSE_MS = 420;
const STARE_MS = 9000; // at or above this she does not blink, she stares
const GAZE_RANGE = 0.14;

export const opticGeometry = (emotion: EmotionKey, size: number): OpticGeometry => {
  const e = EMOTIONS[emotion];
  const bezelRadius = size / 2;
  const irisRadius = (size * e.aperture) / 2;
  return {
    bezelRadius,
    socketRadius: bezelRadius - BEZEL_INSET,
    glowRadius: bezelRadius + GLOW_SPREAD,
    irisRadius,
    pupilRadius: irisRadius * e.pupil,
    lidHeight: size * e.lid,
    lowerLidHeight: size * e.lidB,
    iris: e.iris,
    glowAlpha: e.glow,
    seamAlpha: e.seam,
    blinkMs: e.blinkMs,
    blinkCloseMs: e.slowBlink ? SLOW_BLINK_CLOSE_MS : BLINK_CLOSE_MS,
    blinks: e.blinkMs < STARE_MS,
    tracks: e.track > 0,
    gazeLimit: GAZE_RANGE * e.track,
    showCatchlight: !e.dead,
  };
};
```

- [ ] **Step 6: Build the eye**

A Skia `<Canvas>` of `size × size`, layers outermost-in per spec §6:

1. **Glow** — `<Circle>` at `r = size/2 + 30` with a `<RadialGradient>` in the iris hue at `glow` strength, opacity animated 0.95 → 0.5 on a 2.8s loop.
2. **Bezel** — `<Circle r={size/2}>` in `#191811` with a `<Shadow>` for the inner highlight.
3. **Socket** — `<Circle r={size/2 - 4}>` with a `<RadialGradient>` from `#1c1b15` at 38% to `#0a0a08` at 74%.
4. **Iris rings** — two `<Circle>` strokes at 10% and 20% inset, in the iris hue at 16% and 7% alpha.
5. **Gaze group** — `<Group transform>` driven by the `gaze` shared value, holding:
   - **Iris** — `<Circle r={size * aperture / 2}>` with a `<RadialGradient>` and a 1.5px stroke in the iris hue.
   - **Fibre** — 100 spokes at 3.6° pitch. Draw one `<Path>` of 100 radial lines with a 1.2° arc each, stroked in the iris hue at 26% alpha, group opacity 0.6.
   - **Core** — `<Circle r={irisR * pupil}>` in `#fff` with a `<Shadow>` bloom in the iris hue.
6. **Catchlight** — `<Circle>` in `#ffffffcc` with `<Blur blur={1.4} />` at 25%/20% of the box. Hidden when `dead`.
7. **Lids** — two `<Rect>`s in the bezel colour, top height `lid × size`, bottom height `lidB × size`, both animated.
8. **Wake bloom** — when `waking`, a `<Circle>` with a white-to-iris `<RadialGradient>`, scaling 0.86 → 1.06 while fading, once, over 1.15s.

Emotion transitions use `withTiming` on shared values for `aperture`, `pupil`, `lid`, `lidB` and the iris colour (interpolate via `useDerivedValue`). Blink is a repeating `withRepeat` sequence on the top lid: closed for 90ms every `blinkMs`. `slowBlink` stretches the closure to 420ms and holds. `suspicious` and `bored` never blink — skip the animation entirely when `blinkMs >= 9000`.

Wrap the whole thing in a `<Pressable onPress={onPress}>`.

```ts
// this stays testable. She never shows an error in place of a line.
export const runSpeech = async (deps: SpeechDeps): Promise<SpeechOutcome> => {
  const system = buildSystemPrompt(deps.trigger, deps.lang);
  const user = buildUserPrompt(deps.said, deps.trigger, deps.context);

  try {
    const parsed = parseSpeech(await deps.call(system, user));
    if (parsed) return { ...parsed, authFailed: false };
  } catch (error) {
    if (error instanceof AuthError) {
      const spoken = fallbackFor(deps.trigger, deps.lang);
      return { ...spoken, impression: null, seal: false, authFailed: true };
    }
  }

  const spoken = fallbackFor(deps.trigger, deps.lang);
  return { ...spoken, impression: null, seal: false, authFailed: false };
};
```

```ts
// Specimen-03 is a tamagotchi: hunger drifts UP over time and DOWN when you
// tap. The design doc's prototype only ever decreases — a demo shortcut that
// leaves the specimen fed forever after two taps (spec §12).
export const START_HUNGER = 82;
export const FEED_PER_TAP = 41;
export const HUNGER_FLOOR = 8;
export const DRIFT_PER_HOUR = 2.5;
export const HUNGRY_THRESHOLD = 55;

const clamp = (n: number) => Math.max(0, Math.min(100, n));

export const hungerForNow = (
  storedHunger: number | null,
  lastFedISO: string | null,
  now: Date,
): number => {
  if (storedHunger == null || lastFedISO == null) return START_HUNGER;
  const hours = Math.max(0, (now.getTime() - new Date(lastFedISO).getTime()) / 3_600_000);
  return clamp(storedHunger + hours * DRIFT_PER_HOUR);
};

export const feed = (hunger: number): number =>
  Math.max(HUNGER_FLOOR, Math.min(100, hunger - FEED_PER_TAP));

// Strictly greater, matching the prototype's `hunger > 55`.
export const isHungry = (hunger: number): boolean => hunger > HUNGRY_THRESHOLD;
```

```ts
// src/modules/identity/identity.ts

export interface Identity {
  title: string;
  lines: string[];
  revs: number;
  month: number;
}

// Your declaration of who you intend to be. MEMORY.md is her record of who you
// have been. Neither can edit the other, and she is not told about this one.
export const SEED: Identity = {
  title: 'who I am trying to be',
  lines: [
    'I finish the boring part before I let myself start the interesting part.',
    'I say the awkward thing on the same day I notice it.',
    'I am kind to people who cannot help me.',
    'I do not perform exhaustion to prove the work mattered.',
  ],
  revs: 0,
  month: new Date().getMonth(),
};

const MONTHS = [
  'january', 'february', 'march', 'april', 'may', 'june',
  'july', 'august', 'september', 'october', 'november', 'december',
];

export const stamp = (identity: Identity): string => {
  const times = identity.revs === 0 ? 'not yet revised'
    : identity.revs === 1 ? 'revised once'
      : identity.revs === 2 ? 'revised twice'
        : `revised ${identity.revs} times`;
  return `written in ${MONTHS[identity.month]} · ${times}`;
};

const bump = (identity: Identity, lines: string[]): Identity =>
  ({ ...identity, lines, revs: identity.revs + 1 });

export const editLine = (identity: Identity, index: number, text: string): Identity => {
  const next = text.trim();
  if (next === identity.lines[index]) return identity;
  const lines = [...identity.lines];
  if (next) lines[index] = next;
  else lines.splice(index, 1);
  return bump(identity, lines);
};

export const addLine = (identity: Identity): Identity =>
  ({ ...identity, lines: [...identity.lines, ''] });

export const removeLine = (identity: Identity, index: number): Identity => {
  const lines = [...identity.lines];
  lines.splice(index, 1);
  return bump(identity, lines);
};
```
